**哈希**
查询与数据量无关，O(1)，但大常数
map有伴随数据，set没有

```cpp
#include <iostream>
#include <unordered_set>
#include <unordered_map>
#include <string>
using namespace std;

// 自定义类型做unordered_map的key，必须提供 operator== 和 hash 特化
// （Java里继承Object的hashCode/equals即可，C++不提供则编译报错）
struct Student {
    int age;
    string name;

    Student(int a, string b) : age(a), name(b) {}

    bool operator==(const Student& other) const {
        return age == other.age && name == other.name;
    }
};

template<>
struct std::hash<Student> {
    size_t operator()(const Student& s) const {
        return hash<int>()(s.age) ^ (hash<string>()(s.name) << 1);
    }
};

int main() {
    // C++的string里 == 直接比较值，和Java的equals一样
    string str1 = "Hello";
    string str2 = "Hello";
    // true，C++中 == 就是值比较
    cout << (str1 == str2) << endl;

    unordered_set<string> set;
    set.insert(str1);
    cout << set.count("Hello") << endl;  // 1
    cout << set.count(str2) << endl;     // 1
    set.insert(str2);                    // 重复，不会增加
    cout << set.size() << endl;          // 1
    set.erase(str1);
    set.clear();
    cout << set.empty() << endl;         // 1

    cout << "===========" << endl;

    unordered_map<string, string> map1;
    map1[str1] = "World";
    cout << map1.count("Hello") << endl;           // 1
    cout << map1.count(str2) << endl;              // 1
    cout << map1[str2] << endl;                    // World
    cout << (map1.find("你好") == map1.end()) << endl; // 1（不存在）
    map1.erase("Hello");
    cout << map1.size() << endl;                   // 0
    map1.clear();
    cout << map1.empty() << endl;                  // 1

    cout << "===========" << endl;

    // 笔试中key范围固定可控时，数组可以替代哈希表
    unordered_map<int, int> map2;
    map2[56] = 7285;
    map2[34] = 3671263;
    map2[17] = 716311;
    map2[24] = 1263161;
    // 等价于：
    int arr[100] = {0};
    arr[56] = 7285;
    arr[34] = 3671263;
    arr[17] = 716311;
    arr[24] = 1263161;
    cout << "笔试中哈希表往往会被数组替代" << endl;

    cout << "===========" << endl;
    Student s1(17, "张三");
    Student s2(17, "张三");
    unordered_map<Student, string> map3;
    map3[s1] = "这是张三";
    cout << map3.count(s1) << endl;  // 1
    cout << map3.count(s2) << endl;  // 1，自定义了hash和==所以相等
    map3[s2] = "这是另一个张三";       // 覆盖，因为s1==s2
    cout << map3.size() << endl;     // 1
    cout << map3[s1] << endl;        // 这是另一个张三
    cout << map3[s2] << endl;        // 这是另一个张三
}
```

```cpp
#include <iostream>
#include <map>
#include <set>
#include <queue>
#include <vector>
using namespace std;

int main() {
    // 底层红黑树，按key自动排序
    map<int, string> treeMap;
    treeMap[5] = "这是5";
    treeMap[7] = "这是7";
    treeMap[1] = "这是1";
    treeMap[2] = "这是2";
    treeMap[3] = "这是3";
    treeMap[4] = "这是4";
    treeMap[8] = "这是8";

    cout << treeMap.count(1) << endl;                   // 1
    cout << treeMap.count(10) << endl;                  // 0
    cout << treeMap[4] << endl;                         // 这是4
    treeMap[4] = "张三是4";
    cout << treeMap[4] << endl;                         // 张三是4

    treeMap.erase(4);
    cout << (treeMap.find(4) == treeMap.end()) << endl; // 1（不存在）

    cout << treeMap.begin()->first << endl;    // 1（最小key）
    cout << treeMap.rbegin()->first << endl;   // 8（最大key）
    // floorKey(4)：<= 4 且最近的key
    auto it = treeMap.upper_bound(4);          // 第一个 > 4 的
    if (it != treeMap.begin()) {
        cout << (--it)->first << endl;         // 3（4被删了，<=4最近是3）
    }
    // ceilingKey(4)：>= 4 且最近的key
    it = treeMap.lower_bound(4);               // 第一个 >= 4 的
    if (it != treeMap.end()) {
        cout << it->first << endl;             // 5（4被删了，>=4最近是5）
    }

    cout << "========" << endl;

    set<int> s;
    s.insert(3);
    s.insert(3);    // 重复，不会增加
    s.insert(4);
    s.insert(4);
    cout << "有序表大小 : " << s.size() << endl;   // 2
    while (!s.empty()) {
        cout << *s.begin() << endl;    // 取最小（pollFirst）
        s.erase(s.begin());            // 删最小
        // cout << *s.rbegin() << endl; // 取最大（pollLast）
        // s.erase(prev(s.end()));     // 删最大
    }

    // 堆 —— 注意：C++默认是大根堆！Java默认是小根堆
    priority_queue<int, vector<int>, greater<int>> heap1; // greater = 小根堆
    heap1.push(3);
    heap1.push(3);
    heap1.push(4);
    heap1.push(4);
    cout << "堆大小 : " << heap1.size() << endl;  // 4
    while (!heap1.empty()) {
        cout << heap1.top() << endl;  // 取堆顶
        heap1.pop();                  // 弹出，pop不返回值！
    }

    // 大根堆：默认就是大根堆
    priority_queue<int> heap2;
    heap2.push(3);
    heap2.push(3);
    heap2.push(4);
    heap2.push(4);
    cout << "堆大小 : " << heap2.size() << endl;
    while (!heap2.empty()) {
        cout << heap2.top() << endl;
        heap2.pop();
    }
}
```

```cpp
#include <iostream>
#include <algorithm>
#include <set>
#include <string>
using namespace std;

struct Employee {
    int company;
    int age;

    Employee(int c, int a) : company(c), age(a) {}
};

// 比较器：按年龄升序 —— 返回true表示o1排在o2前
struct EmployeeComparator {
    bool operator()(const Employee& o1, const Employee& o2) const {
        return o1.age < o2.age;
    }
};

int main() {
    Employee arr[] = {
        Employee(2, 27), Employee(1, 60), Employee(4, 19),
        Employee(3, 23), Employee(1, 35), Employee(3, 55)
    };
    int n = sizeof(arr) / sizeof(arr[0]);

    sort(arr, arr + n, EmployeeComparator());
    for (int i = 0; i < n; i++)
        cout << arr[i].company << " , " << arr[i].age << endl;

    cout << "=====" << endl;

    sort(arr, arr + n, [](const Employee& a, const Employee& b) {
        return b.age < a.age;  // 降序：b排在a前即b的优先级更高
    });
    for (int i = 0; i < n; i++)
        cout << arr[i].company << " , " << arr[i].age << endl;

    cout << "=====" << endl;
    // 先按公司编号升序，公司相同按年龄升序
    sort(arr, arr + n, [](const Employee& a, const Employee& b) {
        if (a.company != b.company) return a.company < b.company;
        return a.age < b.age;
    });
    for (int i = 0; i < n; i++)
        cout << arr[i].company << " , " << arr[i].age << endl;

    // set的去重规则：!comp(a,b) && !comp(b,a) 视为相等
    set<Employee, EmployeeComparator> treeSet1;
    for (int i = 0; i < n; i++) treeSet1.insert(arr[i]);
    cout << treeSet1.size() << endl;       // 可能小于6，按age去重

    treeSet1.insert(Employee(2, 27));
    cout << treeSet1.size() << endl;       // 不变，2岁已有过

    cout << "===" << endl;

    // 不想去重 → 比较器增加区分维度
    // Java用toString()做最终区分，C++无等价手段，通用做法：加id字段
    struct EmployeeWithId {
        int company, age, id;
        static int nextId;
        EmployeeWithId(int c, int a) : company(c), age(a), id(nextId++) {}
    };
    int EmployeeWithId::nextId = 0;

    auto cmp = [](const EmployeeWithId& a, const EmployeeWithId& b) {
        if (a.company != b.company) return a.company < b.company;
        if (a.age != b.age) return a.age < b.age;
        return a.id < b.id;  // id确保同公司同龄的对象也不相等
    };
    set<EmployeeWithId, decltype(cmp)> treeSet2(cmp);
    treeSet2.insert(EmployeeWithId(2, 27));
    treeSet2.insert(EmployeeWithId(1, 60));
    treeSet2.insert(EmployeeWithId(4, 19));
    treeSet2.insert(EmployeeWithId(3, 23));
    treeSet2.insert(EmployeeWithId(1, 35));
    treeSet2.insert(EmployeeWithId(3, 55));
    cout << treeSet2.size() << endl;       // 6

    treeSet2.insert(EmployeeWithId(2, 27));
    cout << treeSet2.size() << endl;       // 7，不去重！

    cout << "===" << endl;

    // 字典序
    string str1 = "abcde";
    string str2 = "ks";
    cout << str1.compare(str2) << endl;    // 负数，str1 < str2
    cout << str2.compare(str1) << endl;    // 正数，str2 > str1
}
```

| 概念   | Java           | C++                  |
| ---- | -------------- | -------------------- |
| 哈希集合 | `HashSet<K>`   | `unordered_set<K>`   |
| 哈希映射 | `HashMap<K,V>` | `unordered_map<K,V>` |
| 有序集合 | `TreeSet<K>`   | `set<K>`             |
| 有序映射 | `TreeMap<K,V>` | `map<K>`             |

**unordered_set**
```cpp
#include <unordered_set>

unordered_set<int> s;                      // 空构造
unordered_set<int> s{1,2,3};              // 初始化列表
unordered_set<int> s(vec.begin(), vec.end()); // 迭代器范围构造，常用于去重

// 增
s.insert(x);
s.emplace(args...);

// 删
s.erase(x);          // 按值删除
s.erase(it);         // 按迭代器删除
s.clear();

// 查
s.count(x);          // 是否存在，返回0或1
s.find(x);           // 返回迭代器，未找到返回 s.end()
s.contains(x);       // C++20，判存在

// 遍历
for (auto& v : s) { }

// 其他
s.size();
s.empty();
```

**unordered_map**
```cpp
#include <unordered_map>

unordered_map<string, int> m;              // 空构造
unordered_map<string, int> m{{"a",1},{"b",2}}; // 初始化列表

// 增/改
m["key"] = 42;        // key不存在会插入默认值！谨慎使用
m.insert({"key", 42}); // 已存在则不覆盖
m.emplace("key", 42);

// 删
m.erase("key");
m.erase(it);
m.clear();

// 查
m.at("key");          // key不存在抛异常
m.count("key");       // 是否存在
m.find("key");        // 返回迭代器，未找到返回 m.end()
m.contains("key");    // C++20

// 遍历
for (auto& [k, v] : m) { }  // C++17 结构化绑定

// 其他
m.size();
m.empty();
```

> `m["key"]` 在key不存在时会插入默认值（int默认0），只想查询请用 `find()` 或 `at()`

**set（有序集合）**
```cpp
#include <set>

set<int> s;                   // 空构造，默认升序
s.insert(x);
s.erase(x);                   // 按值删
s.erase(it);                  // 按迭代器删
s.count(x);                   // 是否存在
s.find(x);                    // 返回迭代器，find()==end() 即不存在
s.lower_bound(x);             // 第一个 >= x
s.upper_bound(x);             // 第一个 > x
*s.begin();                   // 最小值
*s.rbegin();                  // 最大值
```

**map（有序映射）**
```cpp
#include <map>

map<int, string> m;
m[k] = v;                     // 增/改
m.erase(k);
m.count(k);                   // 是否存在
m.find(k);                    // != end() 即存在
m.lower_bound(k);             // 第一个 >= k
m.upper_bound(k);             // 第一个 > k
m.begin()->first;             // 最小key
m.rbegin()->first;            // 最大key

// floor: <= k 最近的key
auto it = m.upper_bound(k);
if (it != m.begin()) --it;

// ceiling: >= k 最近的key
auto it = m.lower_bound(k);
if (it != m.end()) ...;
```
