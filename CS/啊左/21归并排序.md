## 递归版
### 代码
```cpp
l = 0;
r = arr.size();
%% 递归整体过程（二分变有序） %%
void mergesort1(int l,int r){
	if(l==r){
		return;
	}%% bacecase %%
	int m = (r+l)/2;
	mergesort1(l,m);
	mergesort1(m+1,r);  %% 流程上这么写，实际最外面的盒子运行到这里一定是已经运行完了 %%
	merge(l,m,r);
}
void merge(int l,int m,int r){
	int a = l;
	int b = m+1; %% 两个指针分别指向两个数组的头 %%
	%% 后面要重写数组，所以要有一个指向新数组的量知道写到哪里 %%
	int i = l;
	vector<int> help[];
	while(a<=m&&b<<r){
		%% 总共两个数组，一个原始的一个help(这两个是完全一样的，这样位置也一样很方便写回)每次到merge的时候要merge的两个内部已经排好了，只需要两个之间互相排了 %%
		help[i++] = arr[a] <= arr[b] ? arr[a++] : arr[b++];
	}
	%% while自带判断，不用if %%
	while(a<=m){
		help[i++] = arr[a++];
	}
	while(b<=r){
		help[i++] = arr[b++];
	}
	%% help只是帮忙排个序，最后还要写回去，不然下次用这次的结果的话不是排好的就不好了 %%
	for(i = l;i<=r;i++){
		arr[i] = help[i];
	}
}
```

## 非递归版
>拿步长1\2\4\8\16···，merge

>递归是有大问题之后层层细分逐步解决。非递归是一开始从小开始做起，后面直接用，按部就班

```cpp
void mergesort2(){
	for(int l,m,r,step = 1;step<n;step<<1){   %% 二分长度->步长（l到m）翻倍->复杂度logn %%
		l = 0;
		while(l<n){   %% 每组for部分都是n，一步步全部都搞 %%
			m = l+step-1;
			if(m+1>=n){
				break;   %% 没有右侧 %%
			}
			%% 如果有右侧 %%
			r = min(l+(step<<1)-1,n-1);
			merge(l,m,r);
			l = r+1;
		}
	}
}
```
## 归并排序为什么块
**比较行为没有浪费**
三傻排序每一次内部的重复都只是为了得到一个数字，，每次还是独立的，浪费
**这里面比较行为被保留了下来变成了有序的部分**
