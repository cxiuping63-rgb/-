## **链表反转**
```CPP
listnode reverselist(listnode head){
	listnode next = NULL;
	listnode pre = NULL;
	
	while(head!=NULL){
		next = head->next;
		head->next = pre;
		pre = head;
		head = next;
	}
	return pre;
}
```
## 合并两个有序链表
```CPP
listnode mergelist(listnode a,listnode b){
	if(a == NULL||b == NULL){
		return a == NULL ? b : a;  %% 这个是要判断是否为空，为空就不用做了 %%
	}
	listnode head = a->val <= b->val ? a : b;  %% 这个要找头节点，直接用，而不是要创建 %%
	listnode cur1 = head->next;
	listnode cur2 = head==a ? b : a;
	listnode HEAD = head;
	while(cur1 != NULL && cur2 != NULL){  %% 有一个是空了就不行 %%
		if(cur1->val <= cur2->val){
			head->next = cur1;
			cur1 = cur1->next;
		}
		else if(cur1->val > cur2->val){
			head->next = cur2;
			cur2 = cur2->next;
		}
		head = head->next;
	}
	%% 剩下的没跳完的只要连上第一个就全好了 %%

	return HEAD;
}
```

## 两个链表相加（个十百千）
```cpp
listnode addlist(listnode a,listnode b){
	listnode cur=NULL,head = NULL;
	int flag = 0
	for(int sum = 0,val = 0;  %% 声明变量 %%
			a != NULL || b != NULL;  %% 判断条件,只有全指向0才可以 %%
			a = a==NULL ? NULL : a->next,
			b = b==NULL ? NULL : b->next){
		sum = (a==null ? 0 : a->val)+(b==null ? 0 : b->val)+flag;
		val = sum % 10;
		flag = sum/10;
		if(head==null){
			head = new listnode(val);  %% listnode(val)这一句相当于__init__ %%
			cur = head;
		}
		else{
			cur->next = new listnode(val);  %% 这里面new相当于在堆里面搞一个，可以跨域保存，不会随意销毁 %%
			cur = cur->next;
		}
	}
	if(flag!=0) cur->next = new listnode(val);
	
	return head;
}
```
