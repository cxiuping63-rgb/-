## 在有序数组里面判断num是否存在
```Cpp
bool ifnum(vector<int>& arr,int num){
	if(arr.empty()){
		return false;
	}
	int l = 0,r = arr.size()-1;
	while(left<=right){
		int mid = (l+r)/2;
		if (arr[mid] == num) {
            return true;
        } 
        else if (arr[mid] < num) {
            left = mid + 1;
        } 
        else {
            right = mid - 1;
        }
	}
	return false;
}
```
## 寻找>=num的最小
```Cpp
bool findmin(vector<int>& a,int b){
	int ans = -1;
	int left = 0,right = a.size()-1;
	while(right>=left){
		int mid = (left+right)/2;   %% l + ((r-1) >> 1)不会溢出 %%
		if(mid >= b){        %% 这里要有等号，因为左边可能有一样的，要全部遍历，不能有落下的，不是找到即可的那种 %%
			right = mid-1;
			ans = mid;
		}
		else if(mid < b){
			left = mid+1;
		}
	}
	return ans;
}
```
