### **用list数据结构**
```cpp
class MyCircularDeque {
private:
    list<int> data;
    int cap;  %% 总体存了多少了 %%

public:
    MyCircularDeque(int k) : cap(k) {}

    bool insertFront(int value) {
        if (data.size() == cap) return false;
        data.push_front(value);
        return true;
    }

    bool insertLast(int value) {
        if (data.size() == cap) return false;
        data.push_back(value);
        return true;
    }

    bool deleteFront() {
        if (data.empty()) return false;
        data.pop_front();
        return true;
    }

    bool deleteLast() {
        if (data.empty()) return false;
        data.pop_back();
        return true;
    }

    int getFront() {
        return data.empty() ? -1 : data.front();
    }

    int getRear() {
        return data.empty() ? -1 : data.back();
    }

    bool isEmpty() { return data.empty(); }
    bool isFull()  { return data.size() == cap; }
};
```
### **不用list数据结构**
```cpp
```cpp
class MyCircularDeque {
private:
    struct Node {
        int val;
        Node *prev, *next;
        Node(int v) : val(v), prev(nullptr), next(nullptr) {}
    };

    Node *head, *tail;
    int cap, sz;

public:
    MyCircularDeque(int k) : cap(k), sz(0) {
        head = new Node(0);
        tail = new Node(0);
        head->next = tail;
        tail->prev = head;
    }

    bool insertFront(int value) {
        if (sz == cap) return false;
        Node* n = new Node(value);
        n->next = head->next;
        n->prev = head;
        head->next->prev = n;
        head->next = n;
        sz++;
        return true;
    }

    bool insertLast(int value) {
        if (sz == cap) return false;
        Node* n = new Node(value);
        n->prev = tail->prev;
        n->next = tail;
        tail->prev->next = n;
        tail->prev = n;
        sz++;
        return true;
    }

    bool deleteFront() {
        if (sz == 0) return false;
        Node* cur = head->next;
        head->next = cur->next;
        cur->next->prev = head;
        delete cur;
        sz--;
        return true;
    }

    bool deleteLast() {
        if (sz == 0) return false;
        Node* cur = tail->prev;
        tail->prev = cur->prev;
        cur->prev->next = tail;
        delete cur;
        sz--;
        return true;
    }

    int getFront() {
        return sz == 0 ? -1 : head->next->val;
    }

    int getRear() {
        return sz == 0 ? -1 : tail->prev->val;
    }

    bool isEmpty() { return sz == 0; }
    bool isFull()  { return sz == cap; }
};
