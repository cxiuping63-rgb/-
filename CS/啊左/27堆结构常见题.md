## 合并K个有序列表
仿照前面的多项式合并（好像是）然后用一个装有三个节点的小根堆。每个弹出之后可以通过next知道下一个节点是什么
```cpp
Listnode mergeKlist(Arraylist<Listnode> arr){
	priority_queue<Listnode,vector<int>,greater<int>> heap;
	for(Listnode h : arr){
		if(h != NULL){
			heap.push(h);
		}
		if(heap == NULL){
			return NULL;
		}
	}
	%% 需要一个头节点（不变），还需要一个一直往下指的指针 %%
	Listnode h = heap.top;
	Listnode p = h;
	if(p.next != NULL){
		heap.push(p.next);
	}
	while(!heap.isempty){
		Listnode cur = heap.top();
		p.next = cur;
		p = cur;
		if(cur.next != NULL){
			heap.push(cur.next);
		}
	}
	return h;    %%  %%
}
```

## 大根堆 & 小根堆 C++ 写法

### 1. 大根堆（默认）

```cpp
// priority_queue 默认就是大根堆，top 是最大值
priority_queue<int> maxHeap;  // 等价于 priority_queue<int, vector<int>, less<int>>

// 插入
maxHeap.push(3);
maxHeap.push(1);
maxHeap.push(5);

cout << maxHeap.top();  // 输出 5（最大值在堆顶）

maxHeap.pop();  // 弹出 5
cout << maxHeap.top();  // 输出 3
```

### 2. 小根堆

```cpp
// 小根堆要传 greater<int>，top 是最小值
priority_queue<int, vector<int>, greater<int>> minHeap;

minHeap.push(3);
minHeap.push(1);
minHeap.push(5);

cout << minHeap.top();  // 输出 1（最小值在堆顶）

minHeap.pop();  // 弹出 1
cout << minHeap.top();  // 输出 3
```

### 3. 从数组构造堆

```cpp
vector<int> arr = {3, 1, 5, 2, 4};

// 方式一：用 make_heap + push_heap（灵活控制大/小根堆）
vector<int> heap = arr;
make_heap(heap.begin(), heap.end());             // 默认大根堆
make_heap(heap.begin(), heap.end(), greater<>()); // 小根堆

// 继续插入元素
heap.push_back(6);
push_heap(heap.begin(), heap.end());

// 弹出堆顶元素
pop_heap(heap.begin(), heap.end());  // 把堆顶换到末尾
int top = heap.back();
heap.pop_back();

// 方式二：priority_queue 直接传迭代器（批量建堆，O(n)）
vector<int> arr2 = {3, 1, 5, 2, 4};
priority_queue<int> maxHeap2(arr2.begin(), arr2.end());
priority_queue<int, vector<int>, greater<int>> minHeap2(arr2.begin(), arr2.end());

// 方式三：手写 heapify（原地建堆，不额外分配内存）
void heapify(vector<int>& arr, int n, int i) {
    int largest = i;
    int left  = 2 * i + 1;
    int right = 2 * i + 2;

    if (left < n && arr[left] > arr[largest])
        largest = left;
    if (right < n && arr[right] > arr[largest])
        largest = right;

    if (largest != i) {
        swap(arr[i], arr[largest]);
        heapify(arr, n, largest);
    }
}

vector<int> arr3 = {3, 1, 5, 2, 4};
int n = arr3.size();
// 从最后一个非叶节点开始向下调整，O(n) 建大根堆
for (int i = n / 2 - 1; i >= 0; --i) {
    heapify(arr3, n, i);
}
// 现在 arr3[0] 是最大值，数组本身就是一个堆
```

### 4. 自定义比较器

```cpp
// 假设堆中存的是 ListNode*，按节点值比较
struct ListNode {
    int val;
    ListNode* next;
    ListNode(int x) : val(x), next(nullptr) {}
};

auto cmp = [](ListNode* a, ListNode* b) { return a->val > b->val; };
priority_queue<ListNode*, vector<ListNode*>, decltype(cmp)> minHeap(cmp);
```

### 核心记忆点
- 大根堆：`priority_queue<int>`，top 最大
- 小根堆：`priority_queue<int, vector<int>, greater<int>>`，top 最小
- 从数组构造：构造函数传迭代器 `(arr.begin(), arr.end())` 即可，时间复杂度 O(n)
- 自定义比较：第三个模板参数放 `decltype(cmp)`，构造函数传 `cmp`
